cmake_minimum_required(VERSION 3.22)
project(qianlinchaguan LANGUAGES C)

set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_custom_target(qlcg_resources ALL
    COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/build/package_resources.py"
    --source "${CMAKE_CURRENT_SOURCE_DIR}" --output "${CMAKE_CURRENT_BINARY_DIR}/assets"
    VERBATIM)

add_library(qlcg_options INTERFACE)
target_include_directories(qlcg_options INTERFACE "${CMAKE_CURRENT_SOURCE_DIR}/src")
if(MSVC)
    set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    target_compile_options(qlcg_options INTERFACE /utf-8 /W3 /we4013 /wd4244 /wd4305 /wd4996 /fp:precise)
    target_compile_definitions(qlcg_options INTERFACE _CRT_SECURE_NO_WARNINGS)
else()
    target_compile_options(qlcg_options INTERFACE -fno-fast-math -ffp-contract=off -Werror=implicit-function-declaration)
endif()

file(GLOB_RECURSE QLCG_SOURCES CONFIGURE_DEPENDS "src/*.c")
list(FILTER QLCG_SOURCES EXCLUDE REGEX "/platform/(win32|android)/")
list(FILTER QLCG_SOURCES EXCLUDE REGEX "/app/app_entry\\.c$")
add_library(qlcg_core STATIC ${QLCG_SOURCES})
target_link_libraries(qlcg_core PUBLIC qlcg_options)
if(WIN32)
    enable_language(RC)
    file(GLOB QLCG_PLATFORM_SOURCES CONFIGURE_DEPENDS "src/platform/win32/*.c")
    add_library(qlcg_platform OBJECT ${QLCG_PLATFORM_SOURCES})
    target_link_libraries(qlcg_platform PUBLIC qlcg_options)
    set(QLCG_SYSTEM_LIBS user32 gdi32 opengl32 ole32 windowscodecs uuid advapi32 imm32)
    add_executable(qianlinchaguan src/app/app_entry.c app.rc)
    target_link_libraries(qianlinchaguan PRIVATE qlcg_core qlcg_platform ${QLCG_SYSTEM_LIBS})
elseif(ANDROID)
    target_compile_options(qlcg_options INTERFACE -fvisibility=hidden -O2)
    if(NOT ANDROID_ABI STREQUAL "arm64-v8a")
        message(FATAL_ERROR "Foundation target requires arm64-v8a")
    endif()
    file(GLOB QLCG_ANDROID_SOURCES CONFIGURE_DEPENDS "src/platform/android/*.c")
    add_library(qlcg_platform STATIC ${QLCG_ANDROID_SOURCES})
    target_link_libraries(qlcg_platform PUBLIC qlcg_options android jnigraphics GLESv3 EGL m)
    message(STATUS "Shared core and EGL backend")
else()
    message(FATAL_ERROR "Supported toolchains: Windows MSVC and Android NDK")
endif()

option(QLCG_ANDROID_GAME "Build GameActivity game" OFF)
if(ANDROID AND QLCG_ANDROID_GAME)
    enable_language(CXX)
    find_package(game-activity REQUIRED CONFIG)
    add_library(qlcg_game SHARED src/platform/android/game_runtime.c)
    set_target_properties(qlcg_game PROPERTIES LINKER_LANGUAGE CXX)
    target_compile_definitions(qlcg_game PRIVATE QLCG_GAME_ACTIVITY)
    target_link_libraries(qlcg_game PRIVATE qlcg_core qlcg_platform game-activity::game-activity_static log)
    target_link_options(qlcg_game PRIVATE "-Wl,-z,max-page-size=16384" "-Wl,-u,GameActivity_onCreate")
endif()
